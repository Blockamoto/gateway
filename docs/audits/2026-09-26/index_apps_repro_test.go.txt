// Audit-only observations loaded with go test -overlay. These tests PASS when

// Graph fixtures follow graph_test.go: synthetic receipts isolate graph/store

func auditIndexSpendFixture(t *testing.T) (*app, blockView, string, string) {
	t.Helper()
	a, genesis := initGraphTestApp(t)
	a.settings.CoreDisabled = true
	a.settings.ServeData = true
	a.cacheWritePending = true // avoid unrelated delayed background cache writes
	if err := a.graphIndexBlock(genesis, true); err != nil {
		t.Fatal(err)
	}
	hash1 := appendGraphTestHeader(t, a, genesisHashDisplay, 111)
	prev := genesis.Transactions[0].TxID
	spendID := strings.Repeat("2", 64)
	v := blockView{
		Height: 1, Hash: hash1, VerificationState: "header_anchored",
		Verification: verificationView{VerifierVersion: blockVerifierVersion, HeaderHash: true, ProofOfWork: true, MerkleRoot: true, TransactionsParsed: true, HeaderChainMatch: true},
		Transactions: []transactionView{{Index: 0, TxID: spendID, Inputs: []inputView{{N: 0, PrevTxID: prev, PrevVout: 0}}, Outputs: []outputView{{N: 0, ValueSats: 1}}}},
	}
	return a, v, prev, spendID
}

func TestAuditObserveGraphDamageBecomesUnspent(t *testing.T) {
	for _, damage := range []string{"delete", "truncate", "crc"} {
		t.Run(damage, func(t *testing.T) {
			a, v, prev, spendID := auditIndexSpendFixture(t)
			if err := a.graphIndexBlock(v, true); err != nil {
				t.Fatal(err)
			}
			if got, ok := a.graphFindSpend(prev, 0); !ok || got.SpendingTxID != spendID {
				t.Fatal("fixture did not commit spend")
			}
			shard, err := graphShardPath(a.graphDir(), "spends", prev)
			if err != nil {
				t.Fatal(err)
			}
			switch damage {
			case "delete":
				err = os.Remove(shard)
			case "truncate":
				err = os.Truncate(shard, 17)
			case "crc":
				var b []byte
				b, err = os.ReadFile(shard)
				if err == nil {
					b[40] ^= 1
					err = os.WriteFile(shard, b, 0600)
				}
			}
			if err != nil {
				t.Fatal(err)
			}
			if _, ok := a.graphFindSpend(prev, 0); ok {
				t.Fatal("damaged record unexpectedly usable")
			}
			got, err := a.resolveSpender(prev, 0)
			if err != nil {
				t.Fatal(err)
			}
			if got.ConfirmedState != "unspent_at_snapshot" || got.Provider != "bod_native_graph" {
				t.Fatalf("baseline defect not observed: %+v", got)
			}
			t.Logf("BASELINE BUG CONFIRMED: %s spend shard => %s, snapshot height %d; expected safe behavior is unknown/repair", damage, got.ConfirmedState, got.Snapshot.Height)
		})
	}
}

func TestAuditObservePrivateGraphKnowledgeServedByBOD(t *testing.T) {
	a, v, prev, spendID := auditIndexSpendFixture(t)
	a.settings.PrivacyMode = true
	v.CacheVisibility = "private" // same marking as the remote path in block.go
	a.indexVerifiedBlockKnowledge(v)
	if _, err := a.cachedSpendLocationForServing(prev, 0); err == nil {
		t.Fatal("fixture must be blocked by cache privacy policy")
	}
	if got, err := a.cachedSpendLocation(prev, 0); err != nil || got.SpendingTxID != spendID {
		t.Fatalf("private knowledge not locally usable: %+v %v", got, err)
	}
	a.source = newOverlayServer(a) // no listening socket: exercise exact BOD handler
	resp, status := a.answerBODRequest(overlayRequest{Type: "spendloc", TxID: prev, Vout: 0})
	if status != "ok" || resp.SpendLocation == nil || resp.SpendLocation.SpendingTxID != spendID {
		t.Fatalf("baseline defect not observed: status=%s response=%+v", status, resp)
	}
	if !intervalsCoverRange(a.buildBODStatus().SpenderCoverage, 1, 1) {
		t.Fatal("expected private-derived height in status")
	}
	t.Log("BASELINE BUG CONFIRMED: private-aware cache refuses serving, but BOD spendloc returns same private-derived graph hit and status advertises its coverage")
}

func TestAuditObserveManifestAdvertisementIgnoresConsent(t *testing.T) {
	exe, err := os.Executable()
	if err != nil {
		t.Fatal(err)
	}
	moduleRoot := filepath.Join(filepath.Dir(exe), "modules")
	if err := os.MkdirAll(moduleRoot, 0700); err != nil {
		t.Fatal(err)
	}
	dir, err := os.MkdirTemp(moduleRoot, "audit-provider-")
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = os.RemoveAll(dir) }) // unique directory inside test binary temp dir
	id := strings.Repeat("a6", 32)
	m := moduleManifest{ID: "audit-unrunnable-private-provider", HostAPI: moduleHostAPIVersion + 999, ProtocolID: id, ProtocolVersion: "1", NetworkAdvertised: false}
	raw, err := json.Marshal(m)
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(filepath.Join(dir, "module.json"), raw, 0600); err != nil {
		t.Fatal(err)
	}
	a := &app{}
	for _, got := range a.extensionCapabilities() {
		if got.ID == id {
			t.Log("BASELINE BUG CONFIRMED: passive host-incompatible network_advertised=false manifest appears in BOD extension capability inventory")
			return
		}
	}
	t.Fatal("baseline defect not observed: manifest was not advertised")
}
