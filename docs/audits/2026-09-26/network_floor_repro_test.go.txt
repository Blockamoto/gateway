func auditSession(t *testing.T, a *app, conn net.Conn, address string) *bitcoinSession {
	t.Helper()
	s := &bitcoinSession{n: a.network, peer: &peerConn{addr: address, conn: conn}, done: make(chan struct{}), slot: make(chan struct{}, 1), replies: make(chan message, 8), serveQueue: make(chan message, 4), satIDs: map[string]bool{}, view: networkPeerView{State: "connected", Address: address, Role: "bitcoin_transport"}}
	a.network.mu.Lock()
	a.network.sessions[address] = s
	a.network.mu.Unlock()
	t.Cleanup(func() { s.close("audit cleanup") })
	return s
}

func TestAuditBaselineRequestTimeoutKillsResponsiveSession(t *testing.T) {
	a := independentTestApp(t)
	local, remote := net.Pipe()
	t.Cleanup(func() { remote.Close() })
	s := auditSession(t, a, local, "127.0.0.1:60111")
	go s.readLoop()
	responsive := make(chan error, 1)
	go func() {
		m, err := readMessage(remote)
		if err != nil || m.command != "getheaders" {
			responsive <- fmt.Errorf("initial request: %s %v", m.command, err)
			return
		}
		if err := writeMessage(remote, "ping", make([]byte, 8)); err != nil {
			responsive <- err
			return
		}
		m, err = readMessage(remote)
		if err == nil && m.command != "pong" {
			err = fmt.Errorf("expected pong, got %s", m.command)
		}
		responsive <- err
	}()
	ctx, cancel := context.WithTimeout(context.Background(), 150*time.Millisecond)
	defer cancel()
	_, err := s.request(ctx, "getheaders", nil, func(m message) bool { return m.command == "headers" })
	if err != context.DeadlineExceeded {
		t.Fatalf("request: %v", err)
	}
	if err := <-responsive; err != nil {
		t.Fatal(err)
	}
	select {
	case <-s.done:
	default:
		t.Fatal("baseline changed: timeout no longer closes session")
	}
	if len(a.network.listSessions()) != 0 {
		t.Fatal("closed session remains registered")
	}
	t.Log("CONFIRMED: withheld headers, successful ping/pong, then request timeout removed otherwise responsive session")
}

func auditBODExchange(pc *peerConn, id string) error {
	pc.conn.SetDeadline(time.Now().Add(3 * time.Second))
	defer pc.conn.SetDeadline(time.Time{})
	env := bodEnvelope{Wire: bodWireVersion, Kind: "request", Type: bodWireType("status"), ID: id, Payload: json.RawMessage(`{"type":"status"}`)}
	raw, _ := json.Marshal(env)
	if err := writeMessage(pc.conn, "bodmsg", raw); err != nil {
		return err
	}
	m, err := readMessage(pc.conn)
	if err != nil {
		return err
	}
	var got bodEnvelope
	if m.command != "bodmsg" || json.Unmarshal(m.payload, &got) != nil || got.ID != id || got.Status != "ok" {
		return fmt.Errorf("unexpected response %s: %s", m.command, m.payload)
	}
	return nil
}

		t.Fatal("baseline changed: inbound appears in managed BOD routing")
	}

}

	s.serveMessage(message{command: "bodmsg", payload: raw})

		t.Fatalf("expected absent reply, got %v", err)
	}

}

func TestAuditBitcoinPoolAutomaticallyReplenishes(t *testing.T) {
	a := independentTestApp(t)
	n := a.network
	n.policy.BitcoinSlots, n.policy.GatewaySlots, n.policy.ScoutSlots = 3, 0, 0
	n.policy.TokenInterval = 20 * time.Millisecond
	for i := 0; i < 6; i++ {
		peer := ordinaryRelay052(t, nil, false)
		n.add(peer.address, "manual_bitcoin", nodeNetworkService|nodeWitnessService)
	}
	n.start()
	eventually051(t, func() bool { return len(n.listSessions()) == 3 })
	first := n.listSessions()[0]
	first.peer.conn.Close()
	eventually051(t, func() bool {
		n.mu.Lock()
		defer n.mu.Unlock()
		return len(n.sessions) == 3 && n.sessions[first.peer.addr] == nil
	})
	active := n.listSessions()
	second, third := active[0], active[1]
	second.peer.conn.Close()
	third.peer.conn.Close()
	eventually051(t, func() bool {
		n.mu.Lock()
		defer n.mu.Unlock()
		return len(n.sessions) == 3 && n.sessions[second.peer.addr] == nil && n.sessions[third.peer.addr] == nil
	})
	t.Log("POSITIVE: automatic one-second maintenance replaced one, then two ordinary sockets from known eligible peers without Retry; test dial token interval shortened to 20ms")
}

func TestAuditWindowsHeaderRollbackCause(t *testing.T) {
	if runtime.GOOS != "windows" {
		t.Skip("Windows-specific append-handle rollback reproduction")
	}
	a := independentTestApp(t)
	raw := mainnetBlockOne(t)
	before, err := os.Stat(a.headersPath)
	if err != nil {
		t.Fatal(err)
	}
	count, batchErr := a.appendCheckedHeaders(context.Background(), a.headersPath, 1, [][]byte{raw[:80], raw[:80]})
	after, err := os.Stat(a.headersPath)
	if err != nil {
		t.Fatal(err)
	}
	t.Logf("production appendCheckedHeaders: before_bytes=%d after_bytes=%d returned_count=%d batch_error=%v", before.Size(), after.Size(), count, batchErr)
	if batchErr == nil || before.Size() != 80 || after.Size() != 160 || count != 1 {
		t.Fatalf("unexpected baseline behavior")
	}
	for _, appendFlag := range []bool{true, false} {
		path := filepath.Join(t.TempDir(), "rollback.bin")
		if err := os.WriteFile(path, make([]byte, 80), 0600); err != nil {
			t.Fatal(err)
		}
		flags := os.O_WRONLY
		if appendFlag {
			flags |= os.O_APPEND
		}
		f, err := os.OpenFile(path, flags, 0600)
		if err != nil {
			t.Fatal(err)
		}
		if !appendFlag {
			if _, err = f.Seek(0, 2); err != nil {
				t.Fatal(err)
			}
		}
		if _, err = f.Write(make([]byte, 80)); err != nil {
			t.Fatal(err)
		}
		truncateErr := f.Truncate(80)
		f.Close()
		st, err := os.Stat(path)
		if err != nil {
			t.Fatal(err)
		}
		t.Logf("minimal native handle: O_APPEND=%v truncate_error=%v final_bytes=%d", appendFlag, truncateErr, st.Size())
		if appendFlag && (truncateErr == nil || st.Size() != 160) {
			t.Fatal("append handle did not reproduce denied rollback")
		}
		if !appendFlag && (truncateErr != nil || st.Size() != 80) {
			t.Fatal("ordinary write handle cannot roll back")
		}
	}
}
